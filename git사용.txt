git 스테이징 : git add.
git main과 브런치 이동 : git switch [main 또는 브런치 이름]
merge : git merge [브런치]
- 만약 main과 브런치에서 동일파일의 같은 부분을 수정하면, conflict 발생



===================================
브랜치 메인으로 푸시하는 방법

# 1. 현재 작업 중인 브랜치에서 변경사항 커밋
git add .
git commit -m "커밋 메시지"

# 2. main 브랜치로 전환
git checkout main

# 3. main 브랜치를 최신 상태로 업데이트 (협업 시 중요)
git pull origin main

# 4. 작업한 브랜치를 main에 merge
git merge 브랜치이름

# 5. 원격 저장소에 push
git push origin main

=============================================
# 1. Git이 설치되어 있나요?
git --version
# git version 2.x.x 이런 식으로 나와야 함

# 2. Git 저장소인가요?
git status
# "fatal: not a git repository" 나오면 안 됨

# 3. 어떤 브랜치에 있나요?
git branch
# 목록이 나와야 함

# 4. 원격 저장소와 연결되어 있나요?
git remote -v
# origin이 나오면 연결됨



# 1. 현재 브랜치 확인
git branch

# 2. 모든 브랜치 확인 (원격 포함)
git branch -a

# 3. 브랜치 생성
git branch 브랜치이름

# 4. 브랜치 생성 + 이동 (추천)
git checkout -b 브랜치이름
# 또는
git switch -c 브랜치이름

# 5. 브랜치 이동
git checkout 브랜치이름
# 또는
git switch 브랜치이름

# 6. 브랜치 삭제
git branch -d 브랜치이름

# 7. 브랜치 강제 삭제
git branch -D 브랜치이름

# 8. 브랜치 이름 변경
git branch -m 기존이름 새이름
